import hashlib
import importlib.util
import json
from pathlib import Path
import sys
import time

worktree = Path('/private/tmp/tool-guardian-cold-profile')
evidence = worktree/'docs/tool-guardian-tuning/evidence'
runner_path = Path('/private/tmp/tool-guardian-profile-checkpoint/scripts/benchmark-high-rate-hooks.py')
sys.path.insert(0,str(runner_path.parent))
spec = importlib.util.spec_from_file_location('targeted_runner',runner_path)
runner = importlib.util.module_from_spec(spec)
sys.modules[spec.name] = runner
spec.loader.exec_module(runner)
def hashes(root):
    paths = set()
    for provider in ('copilot','gemini','codex'):
        directory = root/('.codex/hooks' if provider=='codex' else f'.{provider}/hooks/scripts')
        paths.update(directory.rglob('*.py'))
    for filename in ('hooks/families/tool_guard.py','hooks/manifest.py','scripts/benchmark-high-rate-hooks.py','scripts/benchmark-tool-guard-resources.py','scripts/tool_guard_corpus.py'):
        path = root/filename
        if path.is_file(): paths.add(path)
    return {str(path.relative_to(root)):hashlib.sha256(path.read_bytes()).hexdigest() for path in sorted(paths)}
manifest = {'method':'full147 exact maintained Case/main/invoke; sequential B/C/C/B25samples3warmups; ordinary existing provider cache; fresh log home each case/report; no concurrent probes',
            'runner_sha256':hashlib.sha256(runner_path.read_bytes()).hexdigest(),'worktree_sha256_before':hashes(worktree),'runs':[]}
for index,behavior in enumerate(('baseline','candidate','candidate','baseline'),1):
    root = Path('/private/tmp/tool-guardian-baseline' if behavior=='baseline' else '/private/tmp/tool-guardian-profile-candidate')
    before = hashes(root)
    output = evidence/f'cold-profile-warm-{index}-{behavior}.json'
    sys.argv = [str(runner_path),'--guard-only','--script-root',str(root),'--expected-behavior',behavior,
                '--samples','25','--warmups','3','--output',str(output)]
    status = runner.main()
    after = hashes(root)
    assert status==0 and before==after
    manifest['runs'].append({'order':index,'behavior':behavior,'source_sha256_before':before,'source_sha256_after':after,
                             'provider_bytecode':sorted(str(path.relative_to(root)) for path in root.rglob('*.pyc')),
                             'completed_unix':time.time(),'report_sha256':hashlib.sha256(output.read_bytes()).hexdigest()})
    print('Targeted report complete',index,behavior,flush=True)
manifest['worktree_sha256_after']=hashes(worktree)
assert manifest['worktree_sha256_before']==manifest['worktree_sha256_after']
(evidence/'cold-profile-warm-manifest.json').write_text(json.dumps(manifest,indent=2)+'\n')
