# Adapted from cold-profile-cold-driver.txt. Run only after root GO.
from dataclasses import replace
import runpy
from pathlib import Path
import shutil
import statistics
import tempfile

globals().update(runpy.run_path(str(Path(__file__).with_name('review-repair-common-driver.txt'))))
frozen, roots = configured()
before = state(roots)
cases = {behavior: {case.name: case for case in runner.guard_cases(roots[behavior], behavior)}
         for behavior in ('baseline', 'candidate')}
selected = set(sys.argv[sys.argv.index('--cases') + 1].split(',')) if '--cases' in sys.argv else None
if selected:
    assert selected <= cases['baseline'].keys()
    cases = {behavior: {name: case for name, case in items.items() if name in selected}
             for behavior, items in cases.items()}
report_name = 'review-repair-cold-repeat-25.json' if selected else 'review-repair-cold-25.json'
assert not (evidence / report_name).exists(), 'preserve existing observations'
report = {'method': {'samples_per_condition': 25, 'cases': 147, 'launches': 7350,
                     'order': 'alternating baseline/candidate on even repeats; reverse on odd repeats',
                     'provider_bytecode': 'zero before every fresh-copy launch; ordinary writes allowed after',
                     'logging': 'maintained Case/invoke; fresh home every launch',
                     'budget': 'percase median+5ms/p95+10ms'},
          'identity': frozen['identity'], 'environment': environment(), 'state_before': before,
          'payloads': payloads(roots), 'started_unix': time.time(), 'results': []}
report['method'].update({'cases': len(cases['baseline']), 'launches': len(cases['baseline']) * 50,
                         'exact_targeted_repeat': sorted(selected) if selected else None})
with tempfile.TemporaryDirectory(prefix='guardian-review-cold-') as temporary:
    base = Path(temporary)
    repos = {kind: base / ('repo-' + kind) for kind in {case.repo for case in cases['baseline'].values()}}
    for kind, path in repos.items():
        runner.make_repo(path, kind)
    for case_index, name in enumerate(cases['baseline']):
        samples = {'baseline': [], 'candidate': []}
        row = {'case': name, 'launches': {'baseline': [], 'candidate': []}}
        report['results'].append(row)
        for repeat in range(25):
            order = ('baseline', 'candidate') if repeat % 2 == 0 else ('candidate', 'baseline')
            for launch_order, behavior in enumerate(order):
                original = cases[behavior][name]
                target = base / 'launches' / f'{case_index}-{repeat}-{behavior}'
                target.mkdir(parents=True)
                script = target / 'tool-guard.py'
                sources = {'tool-guard.py': original.script}
                for source in (original.script.parent / 'helpers').rglob('*.py'):
                    sources['helpers/' + str(source.relative_to(original.script.parent / 'helpers'))] = source
                for relative, source in sources.items():
                    destination = target / relative
                    destination.parent.mkdir(parents=True, exist_ok=True)
                    shutil.copy2(source, destination)
                copied = {relative: sha(target / relative) for relative in sources}
                assert copied == {relative: sha(source) for relative, source in sources.items()}
                bytecode = sorted(str(path.relative_to(target)) for path in target.rglob('*.pyc'))
                assert not bytecode
                home = target / 'home'
                home.mkdir()
                case = replace(original, script=script)
                prior_log = log_snapshot(case, home)
                elapsed, status, output = runner.invoke(case, repos[case.repo], home)
                launch = {'repeat': repeat, 'launch_order': launch_order,
                          'provider_bytecode_before': bytecode, 'copied_source_sha256': copied,
                          'copied_source_hashes_verified': True, **observation(case, elapsed, status, output)}
                launch['logging'] = log_evidence(case, prior_log, log_snapshot(case, home))
                row['launches'][behavior].append(launch)
                try:
                    runner.verify_outcome(case, status, output)
                    verify_logging(launch['logging'])
                except Exception as error:
                    launch['verification_failure'] = str(error)
                    write(report_name, report)
                    raise
                samples[behavior].append(elapsed)
                shutil.rmtree(target)
        stats = {behavior: runner.summary(values) for behavior, values in samples.items()}
        median_delta = statistics.median(samples['candidate']) - statistics.median(samples['baseline'])
        p95_delta = runner.percentile(samples['candidate'], .95) - runner.percentile(samples['baseline'], .95)
        row.update({'statistics': stats, 'first_process_ms': {key: values[0] for key, values in samples.items()},
                    'median_delta_ms': median_delta, 'p95_delta_ms': p95_delta,
                    'budget_met': median_delta <= 5 and p95_delta <= 10})
        write(report_name, report)
        if (case_index + 1) % 15 == 0:
            print('Cold cases completed:', case_index + 1, 'of147', flush=True)
report['state_after'] = state(roots)
assert before == report['state_after']
report.update({'sources_unchanged': True, 'completed_unix': time.time(),
               'acceptance': 'PASS' if all(row['budget_met'] for row in report['results']) else 'FAIL'})
write(report_name, report)
print('Cold acceptance:', report['acceptance'], flush=True)
