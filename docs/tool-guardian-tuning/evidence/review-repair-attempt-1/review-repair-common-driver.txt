# Retained external driver only. Run after root GO; never import for preparation.
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import platform
import sys
import time

worktree = Path('/private/tmp/tool-guardian-review-validation')
evidence = worktree / 'docs/tool-guardian-tuning/evidence'
checkpoint = Path('/private/tmp/tool-guardian-profile-checkpoint')
runner_path = checkpoint / 'scripts/benchmark-high-rate-hooks.py'
sys.path.insert(0, str(runner_path.parent))

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def hashes(root):
    paths = set()
    for directory in ('.codex/hooks', '.copilot/hooks/scripts', '.gemini/hooks/scripts'):
        paths.update((root / directory).rglob('*.py'))
    for filename in ('hooks/families/tool_guard.py', 'hooks/manifest.py',
                     'scripts/benchmark-high-rate-hooks.py',
                     'scripts/benchmark-tool-guard-resources.py', 'scripts/tool_guard_corpus.py'):
        path = root / filename
        if path.is_file():
            paths.add(path)
    return {str(path.relative_to(root)): sha(path) for path in sorted(paths)}

def load_module(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    spec.loader.exec_module(module)
    return module

def write(name, value):
    path = evidence / name
    path.write_text(json.dumps(value, indent=2, sort_keys=True) + '\n', encoding='utf-8')
    return sha(path)

def environment():
    return {'platform': platform.platform(), 'machine': platform.machine(),
            'python': sys.version, 'python_executable': sys.executable,
            'python_executable_sha256': sha(Path(sys.executable).resolve()),
            'logging': 'maintained Case/invoke disposable provider-specific writable home',
            'guard_mode': 'block; inherited skip and allowlist removed',
            'os_and_stdlib_cache': 'not cleared',
            'python_cache_settings': {key: os.environ.get(key) for key in
                                     ('PYTHONDONTWRITEBYTECODE', 'PYTHONPYCACHEPREFIX')}}

def state(roots):
    return {'roots': {name: hashes(root) for name, root in roots.items()},
            'checkpoint_scripts': {str(path.relative_to(checkpoint)): sha(path)
                                   for path in sorted((checkpoint / 'scripts').rglob('*.py'))},
            'drivers': {path.name: sha(path) for path in sorted(evidence.glob('review-repair-*-driver.txt'))}}

def configured():
    frozen = json.loads((evidence / 'review-repair-frozen-manifest.json').read_text())
    assert frozen['authorization'] == 'root GO with exclusive timing window'
    assert environment()['python_executable'] == frozen['environment']['python_executable']
    assert environment()['python'] == frozen['environment']['python']
    roots = {name: Path(path) for name, path in frozen['roots'].items()}
    assert state(roots) == frozen['state_after']
    return frozen, roots

driver_bytecode_setting = sys.dont_write_bytecode
sys.dont_write_bytecode = True
try:
    runner = load_module('review_repair_maintained_runner', runner_path)
finally:
    sys.dont_write_bytecode = driver_bytecode_setting

def payloads(roots):
    result = {}
    for behavior in ('baseline', 'candidate'):
        cases = runner.guard_cases(roots[behavior], behavior)
        assert len(cases) == 147
        result[behavior] = {case.name: {'payload_sha256': hashlib.sha256(case.payload).hexdigest(),
                                      'input_bytes': len(case.payload),
                                      'expected_decision': case.expected_decision} for case in cases}
    assert result['baseline'].keys() == result['candidate'].keys()
    for name in result['baseline']:
        assert result['baseline'][name]['payload_sha256'] == result['candidate'][name]['payload_sha256']
    return result

def observation(case, elapsed, status, output):
    return {'elapsed_ms': elapsed, 'exit_code': status, 'output': output,
            'expected_decision': case.expected_decision,
            'actual_decision': runner.decision(case.name.split('.')[0], output),
            'payload_sha256': hashlib.sha256(case.payload).hexdigest()}

def log_snapshot(case, home):
    path = home / 'guard' / 'guard.log' if case.name.startswith('gemini.') else home / 'guard.log'
    data = path.read_bytes() if path.is_file() else b''
    return {'path': str(path), 'exists': path.is_file(), 'size_bytes': len(data),
            'record_count': len(data.splitlines()), 'sha256': hashlib.sha256(data).hexdigest()}

def log_evidence(case, before, after, expected_writes=1):
    exception = case.name in {'copilot.guard.failure', 'gemini.guard.failure', 'codex.guard.failure'}
    if exception:
        try:
            json.loads(case.payload)
        except (ValueError, UnicodeDecodeError):
            pass
        else:
            raise AssertionError('malformed-JSON log exception requires the exact invalid corpus payload')
        expected_writes = 0
    count_delta = after['record_count'] - before['record_count']
    size_delta = after['size_bytes'] - before['size_bytes']
    met = count_delta == expected_writes and (size_delta > 0 if expected_writes else size_delta == 0)
    return {'before': before, 'after': after, 'expected_records_added': expected_writes,
            'records_added': count_delta, 'bytes_added': size_delta,
            'exception': 'fixed malformed-JSON guard.failure returns before logging' if exception else None,
            'required_logging_met': met}

def verify_logging(record):
    assert record['required_logging_met'], 'required guardian log write missing or changed log contract'
