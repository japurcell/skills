# Run after root GO. Default: direct timings. --rss-only: approved native RSS pass.
import re
import runpy
from pathlib import Path
import shlex
import subprocess
import tempfile

globals().update(runpy.run_path(str(Path(__file__).with_name('review-repair-common-driver.txt'))))
frozen, roots = configured()
before = state(roots)
sys.dont_write_bytecode = True
resources = load_module('review_repair_resources', checkpoint / 'scripts/benchmark-tool-guard-resources.py')
sys.dont_write_bytecode = driver_bytecode_setting
rss_only = '--rss-only' in sys.argv
name = ('review-repair-resources-rss-native.json' if '--native' in sys.argv
        else 'review-repair-resources-rss-sandbox.json') if rss_only else 'review-repair-resources-25.json'
assert not (evidence / name).exists(), 'preserve existing observations'
report = {'method': {'samples': 25, 'warmups': 3, 'ceiling_ms': 500,
                     'latency': 'maintained Case/invoke complete Python subprocess',
                     'rss': 'separate native macOS time-l; wrapper excluded from latency'},
          'identity': frozen['identity'], 'environment': environment(),
          'state_before': before, 'results': [], 'started_unix': time.time()}
with tempfile.TemporaryDirectory(prefix='guardian-review-resources-') as temporary:
    base = Path(temporary)
    repo = base / 'repo'
    runner.make_repo(repo, 'clean')
    for provider in resources.PROVIDERS:
        fixtures = resources.fixtures(provider)
        body = 'echo "' + '\ufdfa' * 496 + 'x' * 5 + '"'
        command = 'sh -c ' + shlex.quote(body)
        fixtures += [('nested-normalized-aggregate-max', resources.SHELL_TOOLS[provider], {'command': command}, 'allow'),
                     ('nested-normalized-aggregate-overflow', resources.SHELL_TOOLS[provider], {'command': command + ' '}, 'deny')]
        for fixture, tool, value, expected in fixtures:
            home = base / provider / fixture
            home.mkdir(parents=True)
            data = resources.payload(provider, tool, value)
            case = runner.Case(f'{provider}.resources.{fixture}', resources.script_path(roots['candidate'], provider),
                               data, 'clean', {'GUARD_MODE': 'block'}, expected)
            row = {'case': case.name, 'input_bytes': len(data), 'payload_sha256': hashlib.sha256(data).hexdigest(),
                   'expected_decision': expected, 'observations': []}
            report['results'].append(row)
            if not rss_only:
                for index in range(29):
                    prior_log = log_snapshot(case, home)
                    elapsed, status, output = runner.invoke(case, repo, home)
                    record = {'index': index, **observation(case, elapsed, status, output),
                              'logging': log_evidence(case, prior_log, log_snapshot(case, home))}
                    row['observations'].append(record)
                    try:
                        runner.verify_outcome(case, status, output)
                        verify_logging(record['logging'])
                    except Exception as error:
                        row['verification_failure'] = str(error)
                        write(name, report)
                        raise
                timings = [record['elapsed_ms'] for record in row['observations'][4:]]
                row.update({'first_run_ms': row['observations'][0]['elapsed_ms'], 'samples_ms': timings,
                            'summary': runner.summary(timings), 'rss_pending': True,
                            'ceiling_met': max([row['observations'][0]['elapsed_ms'], *timings]) <= 500})
            else:
                env = os.environ.copy()
                for key in ('SKIP_TOOL_GUARD', 'TOOL_GUARD_ALLOWLIST'):
                    env.pop(key, None)
                env.update({'HOME': str(home), 'XDG_CONFIG_HOME': str(home / '.config'),
                            'XDG_CACHE_HOME': str(home / '.cache'), 'XDG_DATA_HOME': str(home / '.local/share'),
                            'AUDIT_LOG': str(home / 'audit.log'), 'AUDIT_LOCK': str(home / 'audit.lock'),
                            'TOOL_GUARD_LOG_DIR': str(home / 'guard' if provider == 'gemini' else home / 'guard.log'),
                            'SECRETS_LOG_DIR': str(home / 'secrets'),
                            'COPILOT_OBSERVABILITY_LOG_PATH': str(home / 'copilot-observability.ndjson'),
                            'GEMINI_OBSERVABILITY_LOG_PATH': str(home / 'gemini-observability.ndjson'), 'GUARD_MODE': 'block'})
                prior_log = log_snapshot(case, home)
                result = subprocess.run(['/usr/bin/time', '-l', sys.executable, str(case.script)],
                                        input=data, env=env, cwd=repo, capture_output=True, timeout=15)
                row.update({'exit_code': result.returncode, 'stdout': result.stdout.decode('utf-8', 'replace'),
                            'stderr': result.stderr.decode('utf-8', 'replace'),
                            'logging': log_evidence(case, prior_log, log_snapshot(case, home))})
                write(name, report)
                match = re.search(rb'([0-9]+)\s+maximum resident set size', result.stderr)
                if result.returncode or not match:
                    row['rss_failure'] = 'native macOS time-l unavailable or hook failed; retained without acceptance'
                    write(name, report)
                    raise RuntimeError(row['rss_failure'])
                output = json.loads(result.stdout)
                runner.verify_outcome(case, result.returncode, output)
                row.update({'actual_decision': runner.decision(provider, output), 'peak_rss_bytes': int(match.group(1))})
                write(name, report)
                verify_logging(row['logging'])
            write(name, report)
report['state_after'] = state(roots)
assert before == report['state_after']
report.update({'sources_unchanged': True, 'completed_unix': time.time(),
               'acceptance': 'PASS' if rss_only or all(row['ceiling_met'] for row in report['results']) else 'FAIL'})
assert len(report['results']) == 96
write(name, report)
print('Resource acceptance:', report['acceptance'], flush=True)
