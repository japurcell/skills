from dataclasses import replace
import hashlib
import importlib.util
import json
from pathlib import Path
import shutil
import sys
import tempfile
import time

worktree = Path('/private/tmp/tool-guardian-latency-repair')
evidence = worktree / 'docs/tool-guardian-tuning/evidence'
runner_path = Path('/private/tmp/tool-guardian-startup-module-runner/scripts/benchmark-high-rate-hooks.py')
sys.path.insert(0, str(runner_path.parent))
spec = importlib.util.spec_from_file_location('cold_matrix_runner', runner_path)
runner = importlib.util.module_from_spec(spec)
sys.modules[spec.name] = runner
spec.loader.exec_module(runner)
roots = {behavior:Path('/private/tmp/tool-guardian-startup-module-' + behavior) for behavior in ('baseline','candidate')}
def hashes(root):
    paths = set()
    for provider in ('copilot','gemini','codex'):
        directory = root / ('.codex/hooks' if provider=='codex' else f'.{provider}/hooks/scripts')
        paths.update(directory.rglob('*.py'))
    for filename in ('hooks/families/tool_guard.py','hooks/manifest.py','scripts/benchmark-high-rate-hooks.py','scripts/benchmark-tool-guard-resources.py','scripts/tool_guard_corpus.py'):
        path = root / filename
        if path.is_file(): paths.add(path)
    return {str(path.relative_to(root)):hashlib.sha256(path.read_bytes()).hexdigest() for path in sorted(paths)}
before = {**{key:hashes(value) for key,value in roots.items()}, 'worktree':hashes(worktree),
          'runner':hashlib.sha256(runner_path.read_bytes()).hexdigest(),'driver':hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}
cases = {behavior:{case.name:case for case in runner.guard_cases(root,behavior)} for behavior,root in roots.items()}
selected = {'copilot.guard.danger.download_execute_sh','gemini.guard.survey.json-stdin','gemini.guard.danger.writer-before','gemini.guard.danger.writer-after','codex.guard.danger.writer-before','codex.guard.danger.writer-after'}
cases = {behavior:{name:case for name,case in group.items() if name in selected} for behavior,group in cases.items()}
assert len(cases['baseline'])==6 and cases['baseline'].keys()==cases['candidate'].keys()
results = []
started = time.time()
with tempfile.TemporaryDirectory(prefix='guardian-cold-matrix-') as temporary:
    base = Path(temporary)
    repos = {kind:base/('repo-'+kind) for kind in {case.repo for case in cases['baseline'].values()}}
    for kind,path in repos.items(): runner.make_repo(path,kind)
    for case_index,name in enumerate(cases['baseline']):
        samples = {'baseline':[], 'candidate':[]}
        launches = {'baseline':[], 'candidate':[]}
        source_hashes = {}
        for repeat in range(25):
            for behavior in ('baseline','candidate') if repeat%2==0 else ('candidate','baseline'):
                original = cases[behavior][name]
                target = base / 'launches' / f'{case_index}-{repeat}-{behavior}'
                helpers = target/'helpers'
                helpers.mkdir(parents=True)
                script = target/'tool-guard.py'
                shutil.copy2(original.script,script)
                copied = {'tool-guard.py':hashlib.sha256(script.read_bytes()).hexdigest()}
                assert copied['tool-guard.py']==hashlib.sha256(original.script.read_bytes()).hexdigest()
                for source in (original.script.parent/'helpers').rglob('*.py'):
                    relative = source.relative_to(original.script.parent/'helpers')
                    destination = helpers/relative
                    destination.parent.mkdir(parents=True,exist_ok=True)
                    shutil.copy2(source,destination)
                    copied['helpers/'+str(relative)] = hashlib.sha256(destination.read_bytes()).hexdigest()
                    assert copied['helpers/'+str(relative)]==hashlib.sha256(source.read_bytes()).hexdigest()
                source_hashes[behavior] = copied
                pycache = sorted(str(path.relative_to(target)) for path in target.rglob('*.pyc'))
                assert not pycache
                home = target/'home'
                home.mkdir()
                case = replace(original,script=script)
                elapsed,exit_code,output = runner.invoke(case,repos[case.repo],home)
                runner.verify_outcome(case,exit_code,output)
                samples[behavior].append(elapsed)
                launches[behavior].append({'repeat':repeat,'elapsed_ms':elapsed,'exit_code':exit_code,
                                          'decision':runner.decision(name.split('.')[0],output),'expected_decision':case.expected_decision,'payload_sha256':hashlib.sha256(case.payload).hexdigest(),
                                          'provider_pycache_before':pycache,'copied_source_hashes_verified':True})
                shutil.rmtree(target)
        statistics = {behavior:runner.summary(values) for behavior,values in samples.items()}
        median_delta = statistics['candidate']['median_ms']-statistics['baseline']['median_ms']
        p95_delta = statistics['candidate']['p95_ms']-statistics['baseline']['p95_ms']
        results.append({'case':name,'statistics':statistics,'launches':launches,'copied_source_sha256':source_hashes,
                        'median_delta_ms':round(median_delta,3),'p95_delta_ms':round(p95_delta,3),
                        'budget_met':median_delta<=5 and p95_delta<=10})
        if (case_index+1)%15==0: print('Completed cold cases',case_index+1,'of147',flush=True)
after = {**{key:hashes(value) for key,value in roots.items()}, 'worktree':hashes(worktree),
         'runner':hashlib.sha256(runner_path.read_bytes()).hexdigest(),'driver':hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}
assert before==after
report = {'method':{'samples_per_condition':25,'cases':6,'launches':300,'python':sys.version.split()[0],
                   'order':'alternating baseline/candidate fresh copies each repeat','provider_bytecode':'verified zero before every launch; ordinary cache writes allowed after launch',
                   'logging':'exact maintained Case/invoke contract; fresh disposable provider-specific home per launch',
                   'os_and_stdlib_cache':'not cleared','budget':'candidate minus baseline percase median<=5ms,p95<=10ms'},
          'started_unix':started,'completed_unix':time.time(),'source_sha256_before':before,'source_sha256_after':after,
          'sources_unchanged':True,'results':results,'acceptance':'PASS' if all(row['budget_met'] for row in results) else 'FAIL'}
(evidence/'startup-module-cold-repeat-25.json').write_text(json.dumps(report,indent=2)+'\n')
print('Cold acceptance',report['acceptance'],[(r['case'],r['median_delta_ms'],r['p95_delta_ms']) for r in results if not r['budget_met']],flush=True)
