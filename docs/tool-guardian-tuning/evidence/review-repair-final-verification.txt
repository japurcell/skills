def read(name):
    return json.loads((evidence / name).read_text())

def verify_copied_sources(document):
    for case in document['results']:
        provider = case['case'].split('.')[0]
        prefix = '.codex/hooks/' if provider == 'codex' else f'.{provider}/hooks/scripts/'
        for behavior, calls in case['launches'].items():
            source = frozen['state_after']['roots'][behavior]
            expected = {'tool-guard.py': source[prefix + 'tool-guard.py']}
            expected.update({name[len(prefix):]: value for name, value in source.items()
                             if name.startswith(prefix + 'helpers/')})
            for call in calls:
                assert call['copied_source_sha256'] == expected
